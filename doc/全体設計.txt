 4. アルゴリズムの深掘り
 挟み込み判定の完全な手順（6 方向全探索）

 flip_stones(x, y, z, player_color):
     // 入力：盤面内の座標 (x,y,z) と現在のターン color
     // 出力：そのマスに石を置いたとき、挟み込みで反転する石のリスト

     flipped = []                                      // 反転した石の記録用リスト
     directions = [ (+1,0,0), (-1,0,0),              │ ← 6 つの方向ベクトル
                     (0,+1,0), (0,-1,0),
                     (0,0,+1), (0,0,-1) ]

     for each (dx, dy, dz) in directions:
         // ── 探索開始 ───────────────────────────────
         rx ← x + dx                                  │ ← 直後のマス（相手の石か？）
         ry ← y + dy
         rz ← z + dz

         path = []                                    │ ← 挟まれた石の座標を蓄積するリスト

         // ── 「相手の石が続く」のかを確認 ────────────────
         while is_valid(rx, ry, rz):
             cell = board[rx][ry][rz]

             if cell == opponent(player_color):       │ ← 相手の石 → path に追加
                 path.append((rx, ry, rz))
             else if cell == EMPTY:                   │ ← 空マス → この方向は無効
                 break
             else:                                    │ ← 自分の色（または壁）→ 止まる
                 break

         // ── 挟み込みを検出 ────────────────────────────
         if path is not empty:                        │ ← 「相手の石 → 空マス」のケースはスキップ
             tx ← rx + dx                             │ ← 直前のマスのさらに先
             ty ← ry + dy
             tz ← rz + dz

             while is_valid(tx, ty, tz):             │ ← 自分の色を探し続ける
                 cell = board[tx][ty][tz]

                 if cell == player_color:              │ ← 自分の色！→ 挟み込み完了
                     // ── path にある全石を反転 ────────
                     for each (px, py, pz) in path:
                         board[pz][py][px] = player_color
                         flipped.append((px, py, pz))
                     break                            │ ← この方向は終了

                 if cell == EMPTY:                     │ ← 空マス → 挟み込みなし（パス）
                     break

             // else: cell==opponent → さらに先を探索

 有効手の判定ロジック（簡易版：ランダム選択）

 cpp
 std::vector<std::tuple<int,int,int>> CubeBoard::get_valid_moves(int player_color) {
     std::vector<std::tuple<int,int,int>> valid;

     for (int z = 0; z < BOARD_SIZE; ++z) {
         for (int y = 0; y < BOARD_SIZE; ++y) {
             for (int x = 0; x < BOARD_SIZE; ++x) {
                 if (grid_[z][y][x] == EMPTY) {
                     // 挟み込み判定（6 方向探索）
                     auto flipped = flip_stones(x, y, z, player_color);
                     if (!flipped.empty()) {
                         valid.push_back({x, y, z});
                     }
                 }
             }
         }
     }

     return valid;
 }

 AI の評価関数：石数差（depth=3 のミニマックス）

 cpp
 int SimpleAI::evaluate(const CubeBoard& board) const {
     auto [black_count, white_count] = board.count_pieces();

     if (color_ == BLACK) {
         return black_count - white_count;   // 黒が有利なら正値
     } else {
         return white_count - black_count;   // 白が有利なら正値
     }
 }

 std::optional<std::tuple<int,int,int>> SimpleAI::search(const CubeBoard& board) const {
     auto valid_moves = board.get_valid_moves(color_);

     if (valid_moves.empty()) return std::nullopt;  // パス

     int best_score = evaluate(board);
     auto best_move = valid_moves[0];  // 初期化：最初の有効手

     for (const auto& move : valid_moves) {
         board.place_stone(std::get<0>(move), std::get<1>(move), std::get<2>(move), color_);

         int score = minimax(board, depth_ - 1, false);   // 相手のターンに換算

         board.reset();  // バックトラック：盤面を元に戻す

         if (score > best_score) {
             best_score = score;
             best_move = move;
         }
     }

     return best_move;
 }
